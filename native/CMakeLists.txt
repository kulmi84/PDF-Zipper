cmake_minimum_required(VERSION 3.24)
project(PdfZipperShell LANGUAGES CXX)
add_library(PdfZipperShell SHARED Shell.cpp exports.def)
target_compile_features(PdfZipperShell PRIVATE cxx_std_17)
target_compile_definitions(PdfZipperShell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(PdfZipperShell PRIVATE /W4 /WX /utf-8)
target_link_libraries(PdfZipperShell PRIVATE shlwapi shell32 ole32 advapi32)
set_property(TARGET PdfZipperShell PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(ShellTests ShellTests.cpp)
target_compile_features(ShellTests PRIVATE cxx_std_17)
target_compile_definitions(ShellTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ShellTests PRIVATE /W4 /utf-8)
target_link_libraries(ShellTests PRIVATE shell32 ole32)
add_dependencies(ShellTests PdfZipperShell)
add_test(NAME ShellSelection COMMAND ShellTests)
